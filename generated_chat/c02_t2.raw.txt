```python
import math
import cadquery as cq

PARAMS = {
    "W": 900,
    "H": 850,
    "D": 600,
    "leg_h": 100,
    "back_mode": "inset",        # 뒷판을 옆판 사이에 끼우는 인셋
    "top_mode": "none",          # sink base cabinet: no top panel
    "t": 18,
    "back_t": 10,
    "divider_t": 18,
    "door_t": 22,
    "shelf_t": 30,
    "door_gap": 2,
    "door_max_w": 600,
    "n_doors": None,
    "door_layout": None,
    "n_shelves": 0,              # 싱크볼/배관 공간 확보를 위해 선반 없음
    "shelf_min_gap": 320,
    "bottom_clear_min": None,
    "pin_max_spacing": 500,
    "leg_max_spacing": 600,
    "notes": [
        "싱크대 설치용: 상판 없음 (top_mode none). 옆판, 뒷판, 칸막이가 H까지 올라갑니다.",
        "가로 900은 문 2개(칸당 450)로 나누고, 두 문 사이 뒤에 칸막이 1개를 둡니다.",
        "싱크볼과 배관 공간을 위해 선반을 모두 뺐습니다 (n_shelves = 0). 선반 핀도 함께 없습니다.",
        "뒷판은 인셋: 옆판 사이, 바닥판 위부터 캐비닛 상단(H)까지 끼워지며 옆판·바닥판 뒷면과 같은 면입니다.",
        "다리 높이 100 등 언급되지 않은 항목은 회사 기본값을 사용했습니다.",
    ],
}


def _box(x0, x1, y0, y1, z0, z1):
    return (cq.Workplane("XY")
            .box(x1 - x0, y1 - y0, z1 - z0, centered=False)
            .translate((x0, y0, z0)))


def _solid(wp):
    v = wp.val()
    if isinstance(v, cq.Solid):
        return v
    return v.Solids()[0]


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, dt = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t, gap = p["door_t"], p["shelf_t"], p["door_gap"]
    back_mode, top_mode = p["back_mode"], p["top_mode"]
    has_top = top_mode in ("inset", "onset")

    Dc = D - door_t - gap
    pd = (Dc - back_t) if back_mode == "onset" else Dc   # side/top/bottom depth
    inner_d = Dc - back_t                                # dividers, shelves depth

    # ---------- vertical extents ----------
    z0 = leg_h + t                                       # top of bottom panel
    z1 = (H - t) if has_top else H                       # underside of top / carcass top
    clear_h = z1 - z0
    side_top = (H - t) if top_mode == "onset" else H

    # ---------- doors / bays ----------
    layout = p.get("door_layout")
    if layout:
        n = sum(layout)
        cells = list(layout)
    else:
        n = max(1, math.ceil(W / p["door_max_w"]))
        if p.get("n_doors"):
            n = max(n, int(p["n_doors"]))
        cells = [1] * n
    if W / n > p["door_max_w"]:           # never allow a cell wider than the maximum
        n = math.ceil(W / p["door_max_w"])
        cells = [1] * n
    cw = W / n

    # bay boundaries (in door-cell units)
    bounds = []
    acc = 0
    for c in cells[:-1]:
        acc += c
        bounds.append(acc)
    bxs = [j * cw for j in bounds]        # divider centers

    nb = len(cells)
    bays = []
    for k in range(nb):
        xl = t if k == 0 else bxs[k - 1] + dt / 2
        xr = (W - t) if k == nb - 1 else bxs[k] - dt / 2
        lname = "side_left" if k == 0 else "divider_%d" % k
        rname = "side_right" if k == nb - 1 else "divider_%d" % (k + 1)
        bays.append((xl, xr, lname, rname))

    parts = {}   # name -> solid (ordered)
    order = []

    def add(name, solid):
        parts[name] = solid
        order.append(name)

    # ---------- carcass ----------
    panels = {}
    panels["side_left"] = _box(0, t, 0, pd, leg_h, side_top)
    panels["side_right"] = _box(W - t, W, 0, pd, leg_h, side_top)
    for i, xc in enumerate(bxs):
        panels["divider_%d" % (i + 1)] = _box(xc - dt / 2, xc + dt / 2, 0, inner_d, z0, z1)

    other = {}
    other["bottom"] = _box(t, W - t, 0, pd, leg_h, leg_h + t)
    if top_mode == "inset":
        other["top"] = _box(t, W - t, 0, pd, H - t, H)
    elif top_mode == "onset":
        other["top"] = _box(0, W, 0, pd, H - t, H)
    if back_mode == "onset":
        other["back"] = _box(0, W, Dc - back_t, Dc, leg_h, H)
    else:
        other["back"] = _box(t, W - t, Dc - back_t, Dc, z0, z1)

    # ---------- shelves ----------
    if p.get("n_shelves") is not None:
        k = max(0, int(p["n_shelves"]))
        bcm = None
    else:
        bcm = p.get("bottom_clear_min")
        base = max(bcm, p["shelf_min_gap"]) if bcm else p["shelf_min_gap"]
        k = max(0, int(math.floor((clear_h - base) / (p["shelf_min_gap"] + shelf_t))))
    shelf_zs = []
    if k > 0:
        g = (clear_h - k * shelf_t) / (k + 1)
        if bcm and p.get("n_shelves") is None and g < bcm:
            g2 = (clear_h - bcm - k * shelf_t) / k
            for i in range(k):
                shelf_zs.append(z0 + bcm + i * (shelf_t + g2))
        else:
            for i in range(1, k + 1):
                shelf_zs.append(z0 + i * g + (i - 1) * shelf_t)

    pin_r = 2.5
    span = inner_d - 2 * 50
    npin = int(math.ceil(span / p["pin_max_spacing"])) + 1
    pin_ys = [50 + i * span / (npin - 1) for i in range(npin)] if npin > 1 else [50]

    cuts = {}      # panel name -> list of cylinders to cut
    pin_solids = []
    shelf_solids = []
    for bi, (xl, xr, ln, rn) in enumerate(bays):
        for zb in shelf_zs:
            shelf_solids.append(_solid(_box(xl + 1, xr - 1, 0, inner_d, zb, zb + shelf_t)))
            pz = zb - 2.5
            for y in pin_ys:
                # left panel: 8 in the panel, 6 out into the bay
                pl = cq.Solid.makeCylinder(pin_r, 14, cq.Vector(xl - 8, y, pz), cq.Vector(1, 0, 0))
                pin_solids.append(pl)
                cuts.setdefault(ln, []).append(
                    cq.Solid.makeCylinder(pin_r, 8, cq.Vector(xl - 8, y, pz), cq.Vector(1, 0, 0)))
                # right panel
                pr = cq.Solid.makeCylinder(pin_r, 14, cq.Vector(xr - 6, y, pz), cq.Vector(1, 0, 0))
                pin_solids.append(pr)
                cuts.setdefault(rn, []).append(
                    cq.Solid.makeCylinder(pin_r, 8, cq.Vector(xr, y, pz), cq.Vector(1, 0, 0)))

    # apply pin holes
    for name, wp in panels.items():
        for cyl in cuts.get(name, []):
            wp = wp.cut(cq.Workplane("XY").add(cyl))
        panels[name] = wp

    for name in ["side_left", "side_right"]:
        add(name, _solid(panels[name]))
    if "top" in other:
        add("top", _solid(other["top"]))
    add("bottom", _solid(other["bottom"]))
    add("back", _solid(other["back"]))
    for i in range(len(bxs)):
        add("divider_%d" % (i + 1), _solid(panels["divider_%d" % (i + 1)]))
    for i, s in enumerate(shelf_solids):
        add("shelf_%d" % (i + 1), s)
    for i, s in enumerate(pin_solids):
        add("pin_%d" % (i + 1), s)

    # ---------- hinge sides ----------
    hinge = []
    if all(c == 1 for c in cells):
        for i in range(n):
            hinge.append("L" if i % 2 == 0 else "R")
        if n > 1 and n % 2 == 1:
            hinge[-1] = "R"
    else:
        j = 0
        for c in cells:
            if c == 2:
                hinge += ["L", "R"]
            else:
                center = (j + 0.5) * cw
                hinge.append("L" if center <= W / 2 else "R")
            j += c

    # ---------- doors ----------
    dw = cw - 2 * gap
    dh = (H - leg_h) - 2 * gap
    nh = 2 if dh <= 900 else (3 if dh <= 1600 else 4)
    hz_off = [100 + i * (dh - 200) / (nh - 1) for i in range(nh)]
    for i in range(n):
        x0 = i * cw + gap
        zb = leg_h + gap
        door = _box(x0, x0 + dw, -(gap + door_t), -gap, zb, zb + dh)
        cx = x0 + 22.5 if hinge[i] == "L" else x0 + dw - 22.5
        for off in hz_off:
            cup = cq.Solid.makeCylinder(17.5, 12, cq.Vector(cx, -gap, zb + off), cq.Vector(0, -1, 0))
            door = door.cut(cq.Workplane("XY").add(cup))
        add("door_%d" % (i + 1), _solid(door))

    # ---------- legs ----------
    lr = 20
    xa, xb = 30 + lr, W - 30 - lr
    ya, yb = 30 + lr, Dc - 30 - lr
    nx = int(math.ceil((xb - xa) / p["leg_max_spacing"])) + 1
    ny = int(math.ceil((yb - ya) / p["leg_max_spacing"])) + 1
    xs = [xa + i * (xb - xa) / (nx - 1) for i in range(nx)]
    ys = [ya + i * (yb - ya) / (ny - 1) for i in range(ny)]
    li = 1
    for y in ys:
        for x in xs:
            add("leg_%d" % li, cq.Solid.makeCylinder(lr, leg_h, cq.Vector(x, y, 0), cq.Vector(0, 0, 1)))
            li += 1

    return [(name, parts[name]) for name in order]
```
