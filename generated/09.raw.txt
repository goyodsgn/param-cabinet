```python
import math
import cadquery as cq

PARAMS = {
    "W": 3000,
    "H": 850,
    "D": 600,
    "leg_h": 100,
    "back_mode": "onset",
    "t": 18,
    "back_t": 10,
    "divider_t": 18,
    "door_t": 22,
    "shelf_t": 30,
    "door_gap": 2,
    "door_max_w": 600,
    "n_doors": None,
    "n_shelves": None,
    "shelf_min_gap": 320,
    "pin_max_spacing": 500,
    "leg_max_spacing": 600,
    # extra rule values
    "shelf_clearance": 1,
    "pin_d": 5,
    "pin_edge": 50,
    "pin_len": 14,
    "pin_hole_depth": 8,
    "hinge_d": 35,
    "hinge_depth": 12,
    "hinge_edge": 22.5,
    "hinge_end": 100,
    "leg_d": 40,
    "leg_corner_gap": 30,
    "notes": [],
}


def _box(x0, x1, y0, y1, z0, z1):
    return (cq.Workplane("XY")
            .box(x1 - x0, y1 - y0, z1 - z0, centered=False)
            .translate((x0, y0, z0)))


def _cyl_x(x0, length, y, z, d):
    # cylinder along +X starting at x0
    return (cq.Workplane("YZ").workplane(offset=x0)
            .center(y, z).circle(d / 2.0).extrude(length))


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, dt = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t, gap = p["door_t"], p["shelf_t"], p["door_gap"]

    Dc = D - door_t - gap
    onset = (p["back_mode"] == "onset")
    carc_depth = Dc - back_t if onset else Dc   # depth of sides/top/bottom
    inner_depth = Dc - back_t                   # dividers, shelves

    # ---- door count
    n_auto = max(1, math.ceil(round(W / p["door_max_w"], 9)))
    n = n_auto if p["n_doors"] is None else max(int(p["n_doors"]), n_auto)
    cell = W / n

    z0c, z1c = leg_h, H
    parts = {}   # name -> Workplane (panels that may receive holes)
    order = []

    def add(name, wp):
        parts[name] = wp
        order.append(name)

    add("side_left", _box(0, t, 0, carc_depth, z0c, z1c))
    add("side_right", _box(W - t, W, 0, carc_depth, z0c, z1c))
    add("top", _box(t, W - t, 0, carc_depth, z1c - t, z1c))
    add("bottom", _box(t, W - t, 0, carc_depth, z0c, z0c + t))
    if onset:
        add("back", _box(0, W, Dc - back_t, Dc, z0c, z1c))
    else:
        add("back", _box(t, W - t, Dc - back_t, Dc, z0c + t, z1c - t))

    # ---- dividers
    clear_h = (H - leg_h) - 2 * t
    for j in range(1, n):
        xc = j * cell
        add("divider_%d" % j,
            _box(xc - dt / 2.0, xc + dt / 2.0, 0, inner_depth, z0c + t, z1c - t))

    # ---- bays: (left face x, right face x, left panel, right panel)
    bays = []
    for b in range(n):
        if b == 0:
            xl, ln = t, "side_left"
        else:
            xl, ln = b * cell + dt / 2.0, "divider_%d" % b
        if b == n - 1:
            xr, rn = W - t, "side_right"
        else:
            xr, rn = (b + 1) * cell - dt / 2.0, "divider_%d" % (b + 1)
        bays.append((xl, xr, ln, rn))

    # ---- shelf count
    if p["n_shelves"] is None:
        k = max(0, math.floor((clear_h - p["shelf_min_gap"]) /
                              (p["shelf_min_gap"] + shelf_t)))
    else:
        k = max(0, int(p["n_shelves"]))
    space = (clear_h - k * shelf_t) / (k + 1) if k >= 0 else clear_h

    # pin positions along depth
    pe = p["pin_edge"]
    span = inner_depth - 2 * pe
    n_pins = max(1, math.ceil(round(span / p["pin_max_spacing"], 9))) + 1
    pin_ys = [pe + i * span / (n_pins - 1) for i in range(n_pins)]

    pd = p["pin_d"]
    hole_d = p["pin_hole_depth"]
    out_d = p["pin_len"] - hole_d
    sc = p["shelf_clearance"]

    shelves = []
    pins = []
    cuts = []  # (panel name, cylinder wp)
    for (xl, xr, ln, rn) in bays:
        for i in range(k):
            zb = z0c + t + space * (i + 1) + shelf_t * i
            shelves.append(_box(xl + sc, xr - sc, 0, inner_depth, zb, zb + shelf_t))
            zp = zb - pd / 2.0
            for y in pin_ys:
                # left panel: hole into panel (x from xl-8 to xl), pin sticks out
                cuts.append((ln, _cyl_x(xl - hole_d, hole_d, y, zp, pd)))
                pins.append(_cyl_x(xl - hole_d, p["pin_len"], y, zp, pd))
                # right panel
                cuts.append((rn, _cyl_x(xr, hole_d, y, zp, pd)))
                pins.append(_cyl_x(xr - out_d, p["pin_len"], y, zp, pd))

    for name, cyl in cuts:
        parts[name] = parts[name].cut(cyl)

    result = []
    for name in order:
        result.append((name, parts[name].solids().val()))
    for i, s in enumerate(shelves):
        result.append(("shelf_%d" % (i + 1), s.solids().val()))
    for i, s in enumerate(pins):
        result.append(("pin_%d" % (i + 1), s.solids().val()))

    # ---- doors
    dw = cell - 2 * gap
    dh = (H - leg_h) - 2 * gap
    if dh <= 900:
        nh = 2
    elif dh <= 1600:
        nh = 3
    else:
        nh = 4
    hd, hdepth = p["hinge_d"], p["hinge_depth"]
    he, hend = p["hinge_edge"], p["hinge_end"]
    zbot = leg_h + gap
    for i in range(n):
        x0 = i * cell + gap
        door = _box(x0, x0 + dw, -(gap + door_t), -gap, zbot, zbot + dh)
        if i % 2 == 0:
            hinge_left = True
        else:
            hinge_left = False
        if n > 1 and n % 2 == 1 and i == n - 1:
            hinge_left = False
        cx = x0 + he if hinge_left else x0 + dw - he
        for m in range(nh):
            cz = zbot + hend + m * (dh - 2 * hend) / (nh - 1)
            cup = (cq.Workplane("XZ", origin=(0, -gap, 0))
                   .center(cx, cz).circle(hd / 2.0).extrude(hdepth))
            door = door.cut(cup)
        result.append(("door_%d" % (i + 1), door.solids().val()))

    # ---- legs
    ld = p["leg_d"]
    off = p["leg_corner_gap"] + ld / 2.0
    sx = W - 2 * off
    sy = Dc - 2 * off
    nx = max(1, math.ceil(round(sx / p["leg_max_spacing"], 9))) + 1
    ny = max(1, math.ceil(round(sy / p["leg_max_spacing"], 9))) + 1
    idx = 1
    for a in range(nx):
        x = off + a * sx / (nx - 1)
        for b in range(ny):
            y = off + b * sy / (ny - 1)
            leg = (cq.Workplane("XY").center(x, y)
                   .circle(ld / 2.0).extrude(leg_h))
            result.append(("leg_%d" % idx, leg.solids().val()))
            idx += 1

    return result
```
