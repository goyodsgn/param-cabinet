```python
import math
import cadquery as cq

PARAMS = {
    "W": 750,
    "H": 720,
    "D": 550,
    "leg_h": 100,
    "back_mode": "onset",
    "t": 18,
    "back_t": 10,
    "divider_t": 18,
    "door_t": 22,
    "shelf_t": 30,
    "door_gap": 2,
    "door_max_w": 600,
    "n_doors": None,
    "n_shelves": None,
    "shelf_min_gap": 320,
    "pin_max_spacing": 500,
    "leg_max_spacing": 600,
    "hinge_d": 35,
    "hinge_depth": 12,
    "hinge_edge": 22.5,
    "hinge_end": 100,
    "pin_d": 5,
    "pin_len": 14,
    "pin_hole_depth": 8,
    "pin_edge": 50,
    "shelf_clear": 1,
    "leg_d": 40,
    "leg_corner_gap": 30,
    "notes": [],
}


def _box(x, y, z, dx, dy, dz):
    return cq.Workplane("XY").box(dx, dy, dz, centered=False).translate((x, y, z))


def _cyl(r, h, p, d):
    return cq.Solid.makeCylinder(r, h, cq.Vector(*p), cq.Vector(*d))


def _solid(wp):
    obj = wp.val() if isinstance(wp, cq.Workplane) else wp
    if isinstance(obj, cq.Solid):
        return obj
    sols = obj.Solids()
    return sols[0]


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, div_t = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t, gap = p["door_t"], p["shelf_t"], p["door_gap"]
    inset = p["back_mode"] == "inset"

    Dc = D - door_t - gap
    ch = H - leg_h                      # carcass height
    inner_depth = Dc - back_t           # to back panel inner face
    panel_depth = Dc if inset else inner_depth
    z0 = leg_h

    # door count
    n_auto = max(1, math.ceil(W / p["door_max_w"]))
    n = n_auto if p["n_doors"] is None else max(int(p["n_doors"]), n_auto)
    cell = W / n

    parts = []

    # vertical panels (name, x0, x1, z0, height, depth)
    vpanels = [("side_left", 0.0, t)]
    for j in range(1, n):
        c = j * cell
        vpanels.append(("divider_%d" % j, c - div_t / 2, c + div_t / 2))
    vpanels.append(("side_right", W - t, W))

    # shelves
    clear_h = ch - 2 * t
    if p["n_shelves"] is None:
        k = max(0, math.floor((clear_h - p["shelf_min_gap"]) / (p["shelf_min_gap"] + shelf_t)))
    else:
        k = max(0, int(p["n_shelves"]))
    shelf_gap = (clear_h - k * shelf_t) / (k + 1)

    holes = {vp[0]: [] for vp in vpanels}
    shelves = []
    pins = []
    pr = p["pin_d"] / 2.0
    hd = p["pin_hole_depth"]
    pe = p["pin_edge"]
    span = inner_depth - 2 * pe
    n_pins = math.ceil(span / p["pin_max_spacing"]) + 1
    pin_ys = [pe + i * span / (n_pins - 1) for i in range(n_pins)]
    sc = p["shelf_clear"]

    for b in range(len(vpanels) - 1):
        lname, _, bl = vpanels[b]
        rname, br, _ = vpanels[b + 1]
        for s in range(k):
            zb = z0 + t + shelf_gap * (s + 1) + shelf_t * s
            shelves.append(_box(bl + sc, 0, zb, (br - bl) - 2 * sc, inner_depth, shelf_t))
            pz = zb - p["pin_d"] / 2.0  # pin center 2.5 below shelf
            for y in pin_ys:
                # left panel (pin goes +x from inside the panel)
                holes[lname].append(_cyl(pr, hd, (bl - hd, y, pz), (1, 0, 0)))
                pins.append(_cyl(pr, p["pin_len"], (bl - hd, y, pz), (1, 0, 0)))
                # right panel
                holes[rname].append(_cyl(pr, hd, (br, y, pz), (1, 0, 0)))
                pins.append(_cyl(pr, p["pin_len"], (br - (p["pin_len"] - hd), y, pz), (1, 0, 0)))

    # sides and dividers
    for name, x0, x1 in vpanels:
        if name.startswith("side"):
            wp = _box(x0, 0, z0, x1 - x0, panel_depth, ch)
        else:
            wp = _box(x0, 0, z0 + t, x1 - x0, inner_depth, ch - 2 * t)
        for h in holes[name]:
            wp = wp.cut(h)
        parts.append((name, _solid(wp)))

    # top, bottom
    parts.append(("top", _solid(_box(t, 0, H - t, W - 2 * t, panel_depth, t))))
    parts.append(("bottom", _solid(_box(t, 0, z0, W - 2 * t, panel_depth, t))))

    # back
    if inset:
        back = _box(t, Dc - back_t, z0 + t, W - 2 * t, back_t, ch - 2 * t)
    else:
        back = _box(0, Dc - back_t, z0, W, back_t, ch)
    parts.append(("back", _solid(back)))

    # reorder: dividers are already appended in panel loop; shelves & pins
    for i, s in enumerate(shelves):
        parts.append(("shelf_%d" % (i + 1), _solid(s)))
    for i, pn in enumerate(pins):
        parts.append(("pin_%d" % (i + 1), _solid(pn)))

    # doors
    dw = cell - 2 * gap
    dh = ch - 2 * gap
    if dh <= 900:
        nh = 2
    elif dh <= 1600:
        nh = 3
    else:
        nh = 4
    he = p["hinge_end"]
    cup_zs = [he + i * (dh - 2 * he) / (nh - 1) for i in range(nh)]
    for i in range(n):
        x0 = i * cell + gap
        zb = z0 + gap
        door = _box(x0, -(gap + door_t), zb, dw, door_t, dh)
        hinge_left = (i % 2 == 0)
        if n > 1 and n % 2 == 1 and i == n - 1:
            hinge_left = False
        cx = x0 + p["hinge_edge"] if hinge_left else x0 + dw - p["hinge_edge"]
        for cz in cup_zs:
            cup = _cyl(p["hinge_d"] / 2.0, p["hinge_depth"],
                       (cx, -gap, zb + cz), (0, -1, 0))
            door = door.cut(cup)
        parts.append(("door_%d" % (i + 1), _solid(door)))

    # legs
    lr = p["leg_d"] / 2.0
    off = p["leg_corner_gap"] + lr
    xs0, xs1 = off, W - off
    ys0, ys1 = off, Dc - off
    nx = math.ceil((xs1 - xs0) / p["leg_max_spacing"]) + 1
    ny = math.ceil((ys1 - ys0) / p["leg_max_spacing"]) + 1
    idx = 1
    for a in range(nx):
        x = xs0 + (xs1 - xs0) * a / (nx - 1) if nx > 1 else xs0
        for c in range(ny):
            y = ys0 + (ys1 - ys0) * c / (ny - 1) if ny > 1 else ys0
            parts.append(("leg_%d" % idx, _solid(_cyl(lr, leg_h, (x, y, 0), (0, 0, 1)))))
            idx += 1

    return parts
```
