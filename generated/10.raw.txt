```python
import math
import cadquery as cq

PARAMS = {
    "W": 1200,
    "H": 1650,
    "D": 500,
    "leg_h": 80,
    "back_mode": "onset",
    "t": 18,
    "back_t": 10,
    "divider_t": 18,
    "door_t": 22,
    "shelf_t": 30,
    "door_gap": 2,
    "door_max_w": 600,
    "n_doors": None,
    "n_shelves": None,
    "shelf_min_gap": 320,
    "pin_max_spacing": 500,
    "leg_max_spacing": 600,
    "notes": [
        "'다릿발 80mm' was read as leg height leg_h = 80 (leg diameter stays 40).",
        "Tall cabinet: door height is 1566, so each door gets 3 hinge cups.",
        "Door count, shelf count, pins and legs are computed automatically from the dimensions.",
    ],
}


def _box(x0, y0, z0, dx, dy, dz):
    return cq.Workplane("XY").box(dx, dy, dz, centered=False).translate((x0, y0, z0)).val()


def _solid(shape):
    if isinstance(shape, cq.Solid):
        return shape
    sols = shape.Solids()
    return sols[0] if len(sols) == 1 else cq.Solid(cq.Compound.makeCompound(sols).wrapped) if sols else shape


def _cut(solid, cutters):
    s = solid
    for c in cutters:
        s = s.cut(c)
    return _solid(s)


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, div_t = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t, gap = p["door_t"], p["shelf_t"], p["door_gap"]
    door_max_w = p["door_max_w"]

    Dc = D - door_t - gap
    Hc = H - leg_h
    inset = p["back_mode"] == "inset"
    body_d = Dc if inset else Dc - back_t      # depth of sides/top/bottom
    inner_d = Dc - back_t                      # depth of dividers/shelves

    n_auto = max(1, math.ceil(W / door_max_w))
    n = n_auto if p["n_doors"] is None else max(int(p["n_doors"]), n_auto)
    cell = W / n

    parts = []

    # ---------- vertical panels ----------
    clear_h = Hc - 2 * t
    div_xc = [j * cell for j in range(1, n)]

    # shelves count
    if p["n_shelves"] is None:
        k = max(0, math.floor((clear_h - p["shelf_min_gap"]) / (p["shelf_min_gap"] + shelf_t)))
    else:
        k = max(0, int(p["n_shelves"]))
    sgap = (clear_h - k * shelf_t) / (k + 1)
    shelf_bottoms = [leg_h + t + i * sgap + (i - 1) * shelf_t for i in range(1, k + 1)]

    # bays: (left panel inner face x, right panel inner face x, left panel idx, right panel idx)
    # panel idx: 0 = side_left, 1..n-1 = dividers, n = side_right
    bays = []
    edges_l = [t] + [xc + div_t / 2 for xc in div_xc]
    edges_r = [xc - div_t / 2 for xc in div_xc] + [W - t]
    for b in range(n):
        bays.append((edges_l[b], edges_r[b], b, b + 1))

    # pins
    pin_span = inner_d - 2 * 50
    n_pins = math.ceil(pin_span / p["pin_max_spacing"]) + 1
    pin_ys = [50 + i * pin_span / (n_pins - 1) for i in range(n_pins)]

    pin_solids = []
    holes = {i: [] for i in range(n + 1)}
    pr = 2.5
    for (xl, xr, il, ir) in bays:
        for zb in shelf_bottoms:
            zc = zb - 2.5
            for y in pin_ys:
                # left panel: right face at xl, 8 in, 6 out
                pin_solids.append(cq.Solid.makeCylinder(
                    pr, 14, cq.Vector(xl - 8, y, zc), cq.Vector(1, 0, 0)))
                holes[il].append(cq.Solid.makeCylinder(
                    pr, 8, cq.Vector(xl - 8, y, zc), cq.Vector(1, 0, 0)))
                # right panel: left face at xr
                pin_solids.append(cq.Solid.makeCylinder(
                    pr, 14, cq.Vector(xr - 6, y, zc), cq.Vector(1, 0, 0)))
                holes[ir].append(cq.Solid.makeCylinder(
                    pr, 8, cq.Vector(xr, y, zc), cq.Vector(1, 0, 0)))

    side_l = _box(0, 0, leg_h, t, body_d, Hc)
    side_r = _box(W - t, 0, leg_h, t, body_d, Hc)
    parts.append(("side_left", _cut(side_l, holes[0])))
    parts.append(("side_right", _cut(side_r, holes[n])))

    parts.append(("top", _solid(_box(t, 0, H - t, W - 2 * t, body_d, t))))
    parts.append(("bottom", _solid(_box(t, 0, leg_h, W - 2 * t, body_d, t))))

    # back
    if inset:
        back = _box(t, Dc - back_t, leg_h + t, W - 2 * t, back_t, Hc - 2 * t)
    else:
        back = _box(0, Dc - back_t, leg_h, W, back_t, Hc)
    parts.append(("back", _solid(back)))

    # dividers
    for j, xc in enumerate(div_xc, start=1):
        d = _box(xc - div_t / 2, 0, leg_h + t, div_t, inner_d, clear_h)
        parts.append((f"divider_{j}", _cut(d, holes[j])))

    # shelves
    si = 1
    for (xl, xr, il, ir) in bays:
        for zb in shelf_bottoms:
            parts.append((f"shelf_{si}", _solid(_box(xl + 1, 0, zb, (xr - xl) - 2, inner_d, shelf_t))))
            si += 1

    # pins
    for i, ps in enumerate(pin_solids, start=1):
        parts.append((f"pin_{i}", ps))

    # doors
    dw = cell - 2 * gap
    dh = Hc - 2 * gap
    n_h = 2 if dh <= 900 else (3 if dh <= 1600 else 4)
    for i in range(n):
        x0 = i * cell + gap
        z0 = leg_h + gap
        door = _box(x0, -(gap + door_t), z0, dw, door_t, dh)
        hinge_right = (i % 2 == 1)
        if n > 1 and n % 2 == 1 and i == n - 1:
            hinge_right = True
        cx = x0 + dw - 22.5 if hinge_right else x0 + 22.5
        if n_h == 1:
            zs = [z0 + dh / 2]
        else:
            step = (dh - 200) / (n_h - 1)
            zs = [z0 + 100 + m * step for m in range(n_h)]
        cups = [cq.Solid.makeCylinder(17.5, 12, cq.Vector(cx, -gap - 12, z), cq.Vector(0, 1, 0))
                for z in zs]
        parts.append((f"door_{i + 1}", _cut(door, cups)))

    # legs
    leg_r = 20
    x_a, x_b = 30 + leg_r, W - 30 - leg_r
    y_a, y_b = 30 + leg_r, Dc - 30 - leg_r
    nx = math.ceil((x_b - x_a) / p["leg_max_spacing"]) + 1
    ny = math.ceil((y_b - y_a) / p["leg_max_spacing"]) + 1
    xs = [x_a + i * (x_b - x_a) / (nx - 1) for i in range(nx)]
    ys = [y_a + i * (y_b - y_a) / (ny - 1) for i in range(ny)]
    li = 1
    for y in ys:
        for x in xs:
            parts.append((f"leg_{li}", cq.Solid.makeCylinder(
                leg_r, leg_h, cq.Vector(x, y, 0), cq.Vector(0, 0, 1))))
            li += 1

    return parts
```
