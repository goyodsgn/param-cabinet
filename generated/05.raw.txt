```python
import math
import cadquery as cq

PARAMS = {
    # overall size
    "W": 1800.0,
    "H": 1000.0,
    "D": 650.0,
    "leg_h": 100.0,
    "back_mode": "onset",
    # panels
    "t": 18.0,
    "back_t": 10.0,
    "divider_t": 18.0,
    "door_t": 22.0,
    "shelf_t": 30.0,
    # doors
    "door_max_w": 600.0,
    "door_gap": 2.0,
    # hinge cups
    "cup_d": 35.0,
    "cup_depth": 12.0,
    "cup_edge": 22.5,
    "cup_end_offset": 100.0,
    "hinge_h1": 900.0,
    "hinge_h2": 1600.0,
    # shelves
    "shelf_min_gap": 320.0,
    "shelf_side_clear": 1.0,
    # pins
    "pin_d": 5.0,
    "pin_len": 14.0,
    "pin_hole_depth": 8.0,
    "pin_edge": 50.0,
    # legs
    "leg_d": 40.0,
    "leg_corner_gap": 30.0,
    "leg_max_pitch": 600.0,
}


def _box(x0, x1, y0, y1, z0, z1):
    return cq.Solid.makeBox(x1 - x0, y1 - y0, z1 - z0, cq.Vector(x0, y0, z0))


def _cyl(r, h, p, d):
    return cq.Solid.makeCylinder(r, h, cq.Vector(*p), cq.Vector(*d))


def _cut(solid, tools):
    s = solid
    for tool in tools:
        s = s.cut(tool)
    sols = s.Solids()
    return sols[0] if sols else s


def _linspace(a, b, n):
    if n <= 1:
        return [(a + b) / 2.0]
    return [a + (b - a) * i / (n - 1) for i in range(n)]


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, div_t = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t = p["door_t"], p["shelf_t"]
    gap = p["door_gap"]
    mode = p["back_mode"]

    Dc = D - door_t - gap
    Hc = H - leg_h
    z_bot = leg_h
    z_top = H
    inner_depth = Dc - back_t  # depth to the back panel's inner face
    side_depth = inner_depth if mode == "onset" else Dc

    parts = {}
    order = []

    def add(name, solid):
        parts[name] = solid
        order.append(name)

    # ---------- carcass ----------
    add("side_left", _box(0, t, 0, side_depth, z_bot, z_top))
    add("side_right", _box(W - t, W, 0, side_depth, z_bot, z_top))
    add("top", _box(t, W - t, 0, side_depth, z_top - t, z_top))
    add("bottom", _box(t, W - t, 0, side_depth, z_bot, z_bot + t))
    if mode == "onset":
        add("back", _box(0, W, Dc - back_t, Dc, z_bot, z_top))
    else:
        add("back", _box(t, W - t, Dc - back_t, Dc, z_bot + t, z_top - t))

    # ---------- doors count ----------
    n = max(1, math.ceil(W / p["door_max_w"]))
    cell = W / n

    # ---------- inner dividers ----------
    wall_h = Hc - 2 * t
    for j in range(1, n):
        xc = j * cell
        add("divider_%d" % j,
            _box(xc - div_t / 2.0, xc + div_t / 2.0, 0, inner_depth,
                 z_bot + t, z_bot + t + wall_h))

    # ---------- bays ----------
    # vertical panels: (name, left face x, right face x)
    bays = []
    for i in range(n):
        if i == 0:
            left_name, bl = "side_left", t
        else:
            left_name, bl = "divider_%d" % i, i * cell + div_t / 2.0
        if i == n - 1:
            right_name, br = "side_right", W - t
        else:
            right_name, br = "divider_%d" % (i + 1), (i + 1) * cell - div_t / 2.0
        bays.append((left_name, bl, right_name, br))

    # ---------- shelves + pins ----------
    clear_h = Hc - 2 * t
    k = max(0, int(math.floor((clear_h - p["shelf_min_gap"]) /
                              (p["shelf_min_gap"] + shelf_t))))
    space = (clear_h - k * shelf_t) / (k + 1)

    pin_r = p["pin_d"] / 2.0
    hole_d = p["pin_hole_depth"]
    pin_len = p["pin_len"]
    out_len = pin_len - hole_d
    sc = p["shelf_side_clear"]

    shelf_list = []
    pin_list = []
    cuts = {}  # panel name -> list of hole solids

    for (ln, bl, rn, br) in bays:
        for m in range(1, k + 1):
            zb = z_bot + t + m * space + (m - 1) * shelf_t
            shelf_list.append(_box(bl + sc, br - sc, 0, inner_depth, zb, zb + shelf_t))
            zc = zb - pin_r
            for yc in (p["pin_edge"], inner_depth - p["pin_edge"]):
                # left pin: axis along +x, starts inside the left panel
                lp = _cyl(pin_r, pin_len, (bl - hole_d, yc, zc), (1, 0, 0))
                pin_list.append(lp)
                cuts.setdefault(ln, []).append(lp)
                # right pin: starts out in the bay, ends inside right panel
                rp = _cyl(pin_r, pin_len, (br - out_len, yc, zc), (1, 0, 0))
                pin_list.append(rp)
                cuts.setdefault(rn, []).append(rp)

    # drill holes in vertical panels
    for name, tools in cuts.items():
        parts[name] = _cut(parts[name], tools)

    for idx, s in enumerate(shelf_list, 1):
        add("shelf_%d" % idx, s)
    for idx, s in enumerate(pin_list, 1):
        add("pin_%d" % idx, s)

    # ---------- doors ----------
    dw = cell - 2 * gap
    dh = Hc - 2 * gap
    zb_door = leg_h + gap
    if dh <= p["hinge_h1"]:
        hn = 2
    elif dh <= p["hinge_h2"]:
        hn = 3
    else:
        hn = 4
    cup_zs = _linspace(zb_door + p["cup_end_offset"],
                       zb_door + dh - p["cup_end_offset"], hn)
    cup_r = p["cup_d"] / 2.0

    for i in range(n):
        x0 = i * cell + gap
        door = _box(x0, x0 + dw, -(gap + door_t), -gap, zb_door, zb_door + dh)
        hinge_left = (i % 2 == 0)
        if n > 1 and n % 2 == 1 and i == n - 1:
            hinge_left = False
        cx = x0 + p["cup_edge"] if hinge_left else x0 + dw - p["cup_edge"]
        tools = [_cyl(cup_r, p["cup_depth"], (cx, -gap, cz), (0, -1, 0))
                 for cz in cup_zs]
        add("door_%d" % (i + 1), _cut(door, tools))

    # ---------- legs ----------
    off = p["leg_corner_gap"] + p["leg_d"] / 2.0
    span_x = W - 2 * off
    span_y = Dc - 2 * off
    nx = max(1, math.ceil(span_x / p["leg_max_pitch"])) + 1
    ny = max(1, math.ceil(span_y / p["leg_max_pitch"])) + 1
    xs = _linspace(off, W - off, nx)
    ys = _linspace(off, Dc - off, ny)
    li = 1
    for x in xs:
        for y in ys:
            add("leg_%d" % li, _cyl(p["leg_d"] / 2.0, leg_h, (x, y, 0), (0, 0, 1)))
            li += 1

    return [(name, parts[name]) for name in order]
```
