```python
import math
import cadquery as cq

PARAMS = {
    "W": 900,
    "H": 820,
    "D": 600,
    "leg_h": 150,
    "back_mode": "onset",
    "t": 18,
    "back_t": 10,
    "divider_t": 18,
    "door_t": 22,
    "shelf_t": 30,
    "door_gap": 2,
    "door_max_w": 600,
    "n_doors": None,
    "n_shelves": None,
    "shelf_min_gap": 320,
    "pin_max_spacing": 500,
    "leg_max_spacing": 600,
    "notes": [],
    # extra rule values (kept as parameters, not hard-coded in build)
    "hinge_dia": 35,
    "hinge_depth": 12,
    "hinge_edge": 22.5,
    "hinge_end_margin": 100,
    "pin_dia": 5,
    "pin_len": 14,
    "pin_in": 8,
    "pin_edge": 50,
    "shelf_side_clear": 1,
    "leg_dia": 40,
    "leg_edge_gap": 30,
}


def _box(x0, x1, y0, y1, z0, z1):
    return cq.Solid.makeBox(x1 - x0, y1 - y0, z1 - z0, cq.Vector(x0, y0, z0))


def _cyl(d, length, x, y, z, direction):
    return cq.Solid.makeCylinder(d / 2.0, length, cq.Vector(x, y, z), cq.Vector(*direction))


def _solid(shape):
    sols = shape.Solids()
    return sols[0] if len(sols) == 1 else cq.Solid(cq.Compound.makeCompound(sols).wrapped) if sols else shape


def _count(span, max_sp):
    return int(math.ceil(span / max_sp - 1e-9)) + 1


def build(params):
    p = params
    W, H, D = p["W"], p["H"], p["D"]
    leg_h = p["leg_h"]
    t, back_t, dt = p["t"], p["back_t"], p["divider_t"]
    door_t, shelf_t, gap = p["door_t"], p["shelf_t"], p["door_gap"]
    Dc = D - door_t - gap
    Hc = H - leg_h
    z0, z1 = leg_h, H
    inset = p["back_mode"] == "inset"

    # ---- door count
    n = max(1, int(math.ceil(W / p["door_max_w"] - 1e-9)))
    if p["n_doors"] is not None:
        n = max(n, int(p["n_doors"]))
    cell = W / n

    parts = []
    panels = {}   # name -> solid (for later hole cuts)
    order = []

    carc_d = Dc if inset else Dc - back_t   # depth of sides/top/bottom
    inner_d = Dc - back_t                   # depth of dividers & shelves

    panels["side_left"] = _box(0, t, 0, carc_d, z0, z1)
    panels["side_right"] = _box(W - t, W, 0, carc_d, z0, z1)
    panels["top"] = _box(t, W - t, 0, carc_d, z1 - t, z1)
    panels["bottom"] = _box(t, W - t, 0, carc_d, z0, z0 + t)
    if inset:
        panels["back"] = _box(t, W - t, Dc - back_t, Dc, z0 + t, z1 - t)
    else:
        panels["back"] = _box(0, W, Dc - back_t, Dc, z0, z1)

    for j in range(1, n):
        xc = j * cell
        panels["divider_%d" % j] = _box(xc - dt / 2.0, xc + dt / 2.0, 0, inner_d,
                                        z0 + t, z1 - t)

    # ---- bays
    lefts = [t] + [j * cell + dt / 2.0 for j in range(1, n)]
    rights = [j * cell - dt / 2.0 for j in range(1, n)] + [W - t]
    left_panel = ["side_left"] + ["divider_%d" % j for j in range(1, n)]
    right_panel = ["divider_%d" % j for j in range(1, n)] + ["side_right"]

    clear_h = Hc - 2 * t
    if p["n_shelves"] is None:
        k = max(0, int(math.floor((clear_h - p["shelf_min_gap"]) / (p["shelf_min_gap"] + shelf_t))))
    else:
        k = max(0, int(p["n_shelves"]))

    shelves = []
    pins = []
    cuts = {}

    pd = p["pin_dia"]
    pl = p["pin_len"]
    pin_in = p["pin_in"]
    pe = p["pin_edge"]
    sc = p["shelf_side_clear"]
    shelf_d = inner_d
    span = shelf_d - 2 * pe
    n_pins = _count(span, p["pin_max_spacing"])
    pin_ys = [pe + i * span / (n_pins - 1) for i in range(n_pins)]

    if k > 0:
        sp_gap = (clear_h - k * shelf_t) / (k + 1)
        for b in range(n):
            xl, xr = lefts[b], rights[b]
            for i in range(k):
                zb = z0 + t + sp_gap * (i + 1) + shelf_t * i
                shelves.append(_box(xl + sc, xr - sc, 0, shelf_d, zb, zb + shelf_t))
                zc = zb - pd / 2.0
                for y in pin_ys:
                    # left panel of bay
                    pins.append(_cyl(pd, pl, xl - pin_in, y, zc, (1, 0, 0)))
                    cuts.setdefault(left_panel[b], []).append(
                        _cyl(pd, pin_in, xl - pin_in, y, zc, (1, 0, 0)))
                    # right panel of bay
                    pins.append(_cyl(pd, pl, xr - (pl - pin_in), y, zc, (1, 0, 0)))
                    cuts.setdefault(right_panel[b], []).append(
                        _cyl(pd, pin_in, xr, y, zc, (1, 0, 0)))

    # apply holes
    for name, cl in cuts.items():
        s = panels[name]
        for c in cl:
            s = _solid(s.cut(c))
        panels[name] = s

    for nm in ["side_left", "side_right", "top", "bottom", "back"]:
        parts.append((nm, panels[nm]))
    for j in range(1, n):
        parts.append(("divider_%d" % j, panels["divider_%d" % j]))
    for i, s in enumerate(shelves):
        parts.append(("shelf_%d" % (i + 1), s))
    for i, s in enumerate(pins):
        parts.append(("pin_%d" % (i + 1), s))

    # ---- doors
    dw = cell - 2 * gap
    dh = Hc - 2 * gap
    n_h = 2 if dh <= 900 else (3 if dh <= 1600 else 4)
    hd, hdepth = p["hinge_dia"], p["hinge_depth"]
    for i in range(n):
        x0 = i * cell + gap
        zb = z0 + gap
        door = _box(x0, x0 + dw, -(gap + door_t), -gap, zb, zb + dh)
        if n > 1 and n % 2 == 1 and i == n - 1:
            right = True
        else:
            right = (i % 2 == 1)
        cx = x0 + dw - p["hinge_edge"] if right else x0 + p["hinge_edge"]
        m = p["hinge_end_margin"]
        for h in range(n_h):
            cz = zb + m + h * (dh - 2 * m) / (n_h - 1)
            cup = _cyl(hd, hdepth, cx, -gap - hdepth, cz, (0, 1, 0))
            door = _solid(door.cut(cup))
        parts.append(("door_%d" % (i + 1), door))

    # ---- legs
    eg = p["leg_edge_gap"] + p["leg_dia"] / 2.0
    xs0, xs1 = eg, W - eg
    ys0, ys1 = eg, Dc - eg
    nx = _count(xs1 - xs0, p["leg_max_spacing"])
    ny = _count(ys1 - ys0, p["leg_max_spacing"])
    idx = 1
    for ix in range(nx):
        x = xs0 + ix * (xs1 - xs0) / (nx - 1)
        for iy in range(ny):
            y = ys0 + iy * (ys1 - ys0) / (ny - 1)
            parts.append(("leg_%d" % idx, _cyl(p["leg_dia"], leg_h, x, y, 0, (0, 0, 1))))
            idx += 1

    return parts
```
